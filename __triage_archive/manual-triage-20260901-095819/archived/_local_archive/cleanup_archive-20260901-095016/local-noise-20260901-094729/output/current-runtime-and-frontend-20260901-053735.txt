def _is_placeholder_db_value(raw: str | None) -> bool:
    value = str(raw or "").strip()
    return (
        not value
        or "{" in value
        or "}" in value
        or "POSTGRES_" in value
        or value.lower() in {"none", "null", "undefined"}
    )


def _normalize_database_url(raw: str) -> str:
    value = raw.strip()

    if value.startswith("postgresql://"):
        return value.replace("postgresql://", "postgresql+asyncpg://", 1)

    if value.startswith("postgres://"):
        return value.replace("postgres://", "postgresql+asyncpg://", 1)

    if value.startswith("sqlite:///"):
        return value.replace("sqlite:///", "sqlite+aiosqlite:///", 1)

    return value

def _cors_origins() -> list[str]:
    if isinstance(settings.CORS_ORIGINS, list):
        return settings.CORS_ORIGINS
    if isinstance(settings.CORS_ORIGINS, str) and settings.CORS_ORIGINS.strip():
        return [settings.CORS_ORIGINS.strip()]
    return ["http://127.0.0.1:3001", "http://localhost:3001"]

    mounted: list[str] = []

        pref = getattr(api_router, "prefix", "") or ""
        if pref.startswith("/api"):

        else:

        mounted.append("shared.api_router")
    except Exception as e:
        logger.exception("Failed to include shared.api_router")
        skipped.append({"module": "shared.api_router", "error": str(e)})

    module_routers = [
        "app.modules.auth.presentation.router",

    app.state.mounted_routers = mounted
    app.state.skipped_routers = skipped

    logger.info("Mounted routers=%s skipped=%s", mounted, skipped)

    return app

from __future__ import annotations

import hashlib
import logging
import secrets
from uuid import UUID

from app.core.config import settings
from app.modules.users.domain.repositories import UserRepository
from app.shared.application.passwords import PasswordService
from app.shared.infrastructure.email.sender import send_email
from app.shared.infrastructure.security.jwt_service import JwtService

logger = logging.getLogger(__name__)

RESET_PREFIX = "pwdreset:"
RESET_TTL_SECONDS = 60 * 30


def _token_digest(token: str) -> str:
    return hashlib.sha256(token.encode("utf-8")).hexdigest()


class PasswordResetService:
    _mem: dict = {}

    def __init__(
        self,
        user_repo: UserRepository,
        password_service: PasswordService,
        jwt_service: JwtService,
        redis_client=None,
    ):
        self.user_repo = user_repo
        self.password_service = password_service
        self.jwt_service = jwt_service
        self.redis = redis_client

    def _get_redis(self):
        if self.redis is not None:
            return self.redis
        try:
            import redis

            host = getattr(settings.redis, "host", "localhost")
            port = getattr(settings.redis, "port", 6379)
            url = getattr(settings.redis, "url", None) or f"redis://{host}:{port}/0"
            self.redis = redis.from_url(url, decode_responses=True)
            self.redis.ping()
            return self.redis
        except Exception:
            self.redis = None
            return None

    def _store_token(self, token: str, user_id: str) -> None:
        from app.shared.domain.exceptions import ValidationDomainError

        key = RESET_PREFIX + _token_digest(token)
        r = self._get_redis()
        if r is not None:
            r.set(key, user_id, ex=RESET_TTL_SECONDS)
            return

        if settings.is_development:
            logger.warning("Password reset using in-memory fallback storage")
            PasswordResetService._mem[key] = user_id
            return

        logger.error("Password reset unavailable: Redis is not reachable")
        raise ValidationDomainError(
            "Obnova hesla je dočasne nedostupná. Skúste to neskôr."
        )

    def _pop_token(self, token: str) -> str | None:
        key = RESET_PREFIX + _token_digest(token)
        r = self._get_redis()
        if r is not None:
            uid = r.get(key)
            if uid:
                r.delete(key)
            return uid

        if settings.is_development:
            return PasswordResetService._mem.pop(key, None)

        return None

    async def request_reset(self, email: str) -> dict:
        email_n = email.lower().strip()
        user = await self.user_repo.get_by_email(email_n)
        result = {
            "ok": True,
            "message": "Ak účet existuje, poslali sme inštrukcie na e-mail.",
        }
        if user is None:
            return result

        token = secrets.token_urlsafe(32)
        self._store_token(token, str(user.id))

        base = settings.frontend_url.rstrip("/")
        nl = chr(10)
        body = (
            "Obnova hesla CMEP"
            + nl + nl
            + f"Token (platný 30 min): {token}"
            + nl + nl
            + f"Link: {base}/reset-password?token={token}"
            + nl
        )
        sent = send_email(to=email_n, subject="CMEP – obnova hesla", body=body)
        result["emailSent"] = bool(sent)
        if settings.is_development:
            result["resetToken"] = token
        return result

    async def reset_password(self, token: str, new_password: str) -> None:
        from app.shared.domain.exceptions import (
            NotFoundDomainError,
            ValidationDomainError,
        )

        if not token or len(new_password) < 8:
            raise ValidationDomainError(
                "Neplatný token alebo heslo (min. 8 znakov)."
            )
        user_id = self._pop_token(token)
        if not user_id:
            raise ValidationDomainError("Token je neplatný alebo expirovaný.")
        user = await self.user_repo.get_by_id(UUID(str(user_id)))
        if user is None:
            raise NotFoundDomainError("User not found")
        user.password_hash = self.password_service.hash_password(new_password)
        await self.user_repo.update(user)
        await self.jwt_service.revoke_all_user_tokens(str(user.id))

class LoginService:
    def __init__(
        self,
        jwt_service: JwtService,
        user_repo: SqlAlchemyUserRepository,
        password_service: PasswordService,
    ):
        self.jwt_service = jwt_service
        self.user_repo = user_repo
        self.password_service = password_service

    async def execute(self, email: str, password: str) -> AuthTokenPairDTO:
        user = await self.user_repo.get_by_email(email)
        if not user or not self.password_service.verify_password(
            password, user.password_hash
        ):
            raise UnauthorizedDomainError("Invalid credentials")

        subject = str(user.id)
        family_id = str(uuid4())
        access_token, access_expires_at = await self.jwt_service.issue_access_token(
            subject=subject
        )
        refresh_token, refresh_jti, refresh_expires_at = (
            await self.jwt_service.issue_refresh_token(
                subject=subject,
                family_id=family_id,
            )
        )

        expires_in = int(
            (access_expires_at - datetime.now(timezone.utc)).total_seconds()
        )

        return AuthTokenPairDTO(
            access_token=access_token,
            refresh_token=refresh_token,
            token_type="Bearer",
            expires_in=expires_in,
        )


class RefreshTokenService:
    def __init__(self, jwt_service: JwtService, clock: Clock):
        self.jwt_service = jwt_service
        self.clock = clock

    async def execute(self, refresh_token: str) -> AuthTokenPairDTO:
        try:
            payload = await self.jwt_service.decode_and_validate(
                refresh_token, expected_type="refresh"
            )
        except TokenFamilyRevokedError as e:
            logger.warning(f"Replay attack detected: {e}")
            raise UnauthorizedDomainError(
                "Session invalidated due to security concern"
            )
        except TokenUserRevokedError:
            raise UnauthorizedDomainError("Session has been revoked")

        sub = payload.get("sub")
        family_id = payload.get("family_id")
        old_jti = payload.get("jti")
        old_exp = datetime.fromtimestamp(payload.get("exp"), tz=timezone.utc)

        if await self.jwt_service.revocation_store.is_family_revoked(family_id):
            raise UnauthorizedDomainError("Session has been revoked")

        await self.jwt_service.revoke_refresh_token(old_jti, family_id, old_exp)

        access_token, access_expires_at = await self.jwt_service.issue_access_token(
            subject=sub
        )
        new_refresh_token, new_jti, refresh_expires_at = (
            await self.jwt_service.issue_refresh_token(
                subject=sub,
                family_id=family_id,
            )
        )

        expires_in = int(
            (access_expires_at - datetime.now(timezone.utc)).total_seconds()
        )

        return AuthTokenPairDTO(
            access_token=access_token,
            refresh_token=new_refresh_token,
            token_type="Bearer",
            expires_in=expires_in,
        )


class GetCurrentAuthContextService:
    def __init__(
        self,
        jwt_service: JwtService,
        user_repo: SqlAlchemyUserRepository,
        membership_repo,
    ):
        self.jwt_service = jwt_service
        self.user_repo = user_repo
        self.membership_repo = membership_repo

    async def execute(self, access_token: str):
        from uuid import UUID

        from app.modules.auth.application.dto import CurrentAuthContextDTO
        from app.modules.users.application.mappers import to_user_dto

        try:
            payload = await self.jwt_service.decode_and_validate(
                access_token, expected_type="access"
            )
        except TokenExpiredError:
            raise UnauthorizedDomainError("Token has expired")
        except TokenRevokedError:
            raise UnauthorizedDomainError("Token has been revoked")
        except TokenUserRevokedError:
            raise UnauthorizedDomainError("Session has been revoked")
        except InvalidTokenError as e:
            raise UnauthorizedDomainError(f"Invalid token: {e.message}")

        sub = payload.get("sub")
        user = await self.user_repo.get_by_id(UUID(sub))
        if user is None:
            raise UnauthorizedDomainError("User not found")

        memberships = await self.membership_repo.list_by_user_id(UUID(sub))

        role_codes = sorted({m.membership_role.value for m in memberships})

        membership_dicts = [
            {
                "id": m.id,
                "companyId": m.company_id,
                "membershipRole": m.membership_role.value,
                "status": m.status.value,
                "isPrimaryContact": m.is_primary_contact,
            }
            for m in memberships
        ]

        return CurrentAuthContextDTO(
            user=to_user_dto(user),
            role_codes=role_codes,
            memberships=membership_dicts,
        )


class LogoutService:
    def __init__(self, jwt_service: JwtService):
        self.jwt_service = jwt_service

    async def execute(self, refresh_token: str | None = None) -> None:
        if refresh_token:
            try:
                refresh_payload = await self.jwt_service.decode_and_validate(
                    refresh_token, expected_type="refresh"
                )
                refresh_jti = refresh_payload.get("jti")
                refresh_family = refresh_payload.get("family_id")
                refresh_exp = datetime.fromtimestamp(
                    refresh_payload.get("exp"), tz=timezone.utc
                )
                await self.jwt_service.revoke_refresh_token(
                    refresh_jti, refresh_family, refresh_exp
                )
            except Exception as e:
                logger.debug(f"Could not revoke refresh token: {e}")

        logger.info("User logged out successfully")

        self.session = session

    async def add(self, session_entity: RefreshSession) -> None:
        self.session.add(to_model(session_entity))
        await self.session.commit()

    async def get_by_jti(self, jti: str) -> RefreshSession | None:
        stmt = select(RefreshSessionModel).where(RefreshSessionModel.refresh_token_jti == jti)
        result = await self.session.execute(stmt)
        model = result.scalar_one_or_none()
        return to_domain(model) if model else None

    async def get_active_by_user_id(self, user_id: UUID) -> list[RefreshSession]:
        stmt = select(RefreshSessionModel).where(
            RefreshSessionModel.user_id == str(user_id),
            RefreshSessionModel.revoked_at.is_(None),
        )
        result = await self.session.execute(stmt)
        return [to_domain(item) for item in result.scalars().all()]

    async def update(self, session_entity: RefreshSession) -> None:
        model = await self.session.get(RefreshSessionModel, str(session_entity.id))
        if model is None:
            return
        model.revoked_at = session_entity.revoked_at
        model.replaced_by_jti = session_entity.replaced_by_jti
        model.reuse_detected_at = session_entity.reuse_detected_at
        model.updated_at = session_entity.updated_at
        model.updated_by_user_id = str(session_entity.updated_by_user_id) if session_entity.updated_by_user_id else None
        await self.session.commit()

    async def revoke_family(self, family_id: str, revoked_at) -> None:
        stmt = (
            update(RefreshSessionModel)
            .where(RefreshSessionModel.token_family_id == family_id, RefreshSessionModel.revoked_at.is_(None))

        )
        await self.session.execute(stmt)
        await self.session.commit()

"""Token revocation store — Redis or in-memory (Termux / dev)."""
from __future__ import annotations

from datetime import datetime, timedelta, timezone
from typing import Any

from app.core.config import settings

REFRESH_TOKEN_TTL_DAYS = int(str(settings.jwt.refresh_token_expire_days))


class _MemoryBackend:
    def __init__(self) -> None:
        self._store: dict[str, tuple[str, float | None]] = {}

    def _purge(self) -> None:
        now = datetime.now(timezone.utc).timestamp()
        dead = [k for k, (_, exp) in self._store.items() if exp is not None and exp <= now]
        for k in dead:
            self._store.pop(k, None)

    async def set(self, key: str, value: str, ex: int | None = None) -> None:
        self._purge()
        exp = None
        if ex is not None and ex > 0:
            exp = datetime.now(timezone.utc).timestamp() + ex
        self._store[key] = (value, exp)

    async def get(self, key: str) -> str | None:
        self._purge()
        item = self._store.get(key)
        if not item:
            return None
        return item[0]

    async def exists(self, key: str) -> int:
        self._purge()
        return 1 if key in self._store else 0

    async def close(self) -> None:
        self._store.clear()


class TokenRevocationStore:
    def __init__(self, redis_url: str | None = None):
        self.redis_url = redis_url or str(settings.redis.url)
        self._client: Any = None
        self._use_memory = (
            not getattr(settings.redis, "enabled", False)
            or self.redis_url.startswith("memory:")
            or self.redis_url in ("", "none", "disabled")
        )

    async def _get_client(self) -> Any:
        if self._client is not None:
            return self._client
        if self._use_memory:
            self._client = _MemoryBackend()
            return self._client
        try:
            import redis.asyncio as redis
            self._client = redis.from_url(self.redis_url, decode_responses=True)
            return self._client
        except Exception:
            self._client = _MemoryBackend()
            self._use_memory = True
            return self._client

    async def revoke_access_token(self, jti: str, exp: datetime) -> None:
        client = await self._get_client()
        ttl = int((exp - datetime.now(timezone.utc)).total_seconds())
        if ttl > 0:
            await client.set(f"revoked:access:{jti}", "1", ex=ttl)

    async def revoke_refresh_token(self, jti: str, family_id: str, exp: datetime) -> None:
        client = await self._get_client()
        ttl = int((exp - datetime.now(timezone.utc)).total_seconds())
        if ttl > 0:
            await client.set(f"revoked:refresh:{jti}", "1", ex=ttl)
            await client.set(f"revoked:family:{family_id}", "1", ex=ttl)

    async def revoke_user_tokens_after(self, subject: str, cutoff: datetime) -> None:
        client = await self._get_client()
        cutoff_utc = cutoff.astimezone(timezone.utc)
        ttl = int(timedelta(days=REFRESH_TOKEN_TTL_DAYS).total_seconds())
        await client.set(
            f"revoked:user:{subject}:after",
            cutoff_utc.isoformat(),
            ex=ttl,
        )

    async def get_user_tokens_revoked_after(self, subject: str) -> datetime | None:
        client = await self._get_client()
        value = await client.get(f"revoked:user:{subject}:after")
        if not value:
            return None
        try:
            return datetime.fromisoformat(value)
        except Exception:
            return None

    async def is_access_token_revoked(self, jti: str) -> bool:
        client = await self._get_client()
        try:
            return await client.exists(f"revoked:access:{jti}") > 0
        except Exception:
            return False

    async def is_family_revoked(self, family_id: str) -> bool:
        client = await self._get_client()
        try:
            return await client.exists(f"revoked:family:{family_id}") > 0
        except Exception:
            return False

    async def close(self) -> None:
        if self._client is not None:
            close = getattr(self._client, "close", None)
            if close:
                await close()
            self._client = None


_revocation_store: TokenRevocationStore | None = None


def get_revocation_store() -> TokenRevocationStore:
    global _revocation_store
    if _revocation_store is None:
        _revocation_store = TokenRevocationStore()
    return _revocation_store

from datetime import datetime, timedelta, timezone
from typing import Any
from uuid import uuid4

import jwt

from app.core.config import settings
from app.shared.infrastructure.security.jwt_exceptions import (
    InvalidTokenError,
    TokenExpiredError,
    TokenFamilyRevokedError,
    TokenRevokedError,
    TokenTypeMismatchError,
    TokenUserRevokedError,
)
from app.shared.infrastructure.security.jwt_revocation_store import (
    TokenRevocationStore,
    get_revocation_store,
)

JWT_SECRET = settings.jwt.secret_key
JWT_ALGORITHM = settings.jwt.algorithm
ACCESS_TOKEN_TTL_MINUTES = int(os.getenv("ACCESS_TOKEN_TTL_MINUTES", "15"))
REFRESH_TOKEN_TTL_DAYS = int(str(settings.jwt.refresh_token_expire_days))


class JwtService:
    def __init__(self, revocation_store: TokenRevocationStore | None = None):
        self._revocation_store = revocation_store
        self._owns_store = revocation_store is None

    @property
    def revocation_store(self) -> TokenRevocationStore:
        if self._revocation_store is None:
            self._revocation_store = get_revocation_store()
        return self._revocation_store

    async def issue_access_token(
        self, subject: str, extra_claims: dict[str, Any] | None = None
    ) -> tuple[str, datetime]:
        issued_at = datetime.now(timezone.utc)
        expires_at = issued_at + timedelta(minutes=ACCESS_TOKEN_TTL_MINUTES)
        jti = str(uuid4())
        payload: dict[str, Any] = {
            "sub": subject,
            "type": "access",
            "jti": jti,
            "iat": issued_at,
            "exp": expires_at,
        }
        if extra_claims:
            payload.update(extra_claims)
        return jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM), expires_at

    async def issue_refresh_token(
        self, subject: str, family_id: str, extra_claims: dict[str, Any] | None = None
    ) -> tuple[str, str, datetime]:
        issued_at = datetime.now(timezone.utc)
        expires_at = issued_at + timedelta(days=REFRESH_TOKEN_TTL_DAYS)
        jti = str(uuid4())
        payload: dict[str, Any] = {
            "sub": subject,
            "type": "refresh",
            "jti": jti,
            "family_id": family_id,
            "iat": issued_at,
            "exp": expires_at,
        }
        if extra_claims:
            payload.update(extra_claims)
        return jwt.encode(payload, JWT_SECRET, algorithm=JWT_ALGORITHM), jti, expires_at

    async def decode_and_validate(
        self, token: str, expected_type: str = "access"
    ) -> dict[str, Any]:
        """
        Decode and validate token with full security checks.
        - Verifies signature
        - Checks expiry
        - Checks revocation list
        - Validates token type
        """
        try:
            payload = jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        except jwt.ExpiredSignatureError:
            raise TokenExpiredError("Token has expired")
        except jwt.InvalidTokenError as e:
            raise InvalidTokenError(f"Invalid token: {str(e)}")

        token_type = payload.get("type")
        if token_type != expected_type:
            raise TokenTypeMismatchError(
                f"Expected {expected_type} token, got {token_type}"
            )

        sub = payload.get("sub")
        iat_raw = payload.get("iat")
        if sub and iat_raw is not None:
            token_issued_at = datetime.fromtimestamp(iat_raw, tz=timezone.utc)
            revoked_after = await self.revocation_store.get_user_tokens_revoked_after(sub)
            if revoked_after is not None and token_issued_at <= revoked_after:
                raise TokenUserRevokedError(
                    "All existing user sessions have been revoked"
                )

        if expected_type == "access":
            jti = payload.get("jti")
            if await self.revocation_store.is_access_token_revoked(jti):
                raise TokenRevokedError("Access token has been revoked")

        if expected_type == "refresh":
            family_id = payload.get("family_id")
            if await self.revocation_store.is_family_revoked(family_id):
                raise TokenFamilyRevokedError(
                    "Token family revoked - possible replay attack"
                )

        return payload

    async def revoke_access_token(self, jti: str, exp: datetime) -> None:
        await self.revocation_store.revoke_access_token(jti, exp)

    async def revoke_refresh_token(self, jti: str, family_id: str, exp: datetime) -> None:
        await self.revocation_store.revoke_refresh_token(jti, family_id, exp)

    async def revoke_all_user_tokens(
        self, subject: str, cutoff: datetime | None = None
    ) -> None:
        await self.revocation_store.revoke_user_tokens_after(
            subject,
            cutoff or datetime.now(timezone.utc),
        )

    def decode(self, token: str) -> dict[str, Any]:
        return jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])

import { tokenStore } from '../auth/token-store';

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const baseUrl = process.env.NEXT_PUBLIC_API_BASE_URL || 'http://localhost:8001';
  const url = `${baseUrl}${path.startsWith('/') ? path : '/' + path}`;
  const token = tokenStore.getToken();

  const headers = new Headers({
    'Content-Type': 'application/json',
    ...(options.headers || {}),
  });

  if (token) {
    headers.set('Authorization', `Bearer ${token}`);
  }

  const response = await fetch(url, { ...options, headers });

  if (!response.ok) {
    if (response.status === 401) {
      tokenStore.clearToken();
      if (typeof window !== 'undefined') window.location.href = '/login';
    }
    const errorData = await response.json().catch(() => ({}));
    throw {
      message: errorData.detail || errorData.message || 'An unexpected error occurred',
      status: response.status,
      errors: errorData.errors,
    };
  }
  if (response.status === 204) return {} as T;
  return response.json();
}

export const http = {

  delete: <T>(path: string, options?: RequestInit) => request<T>(path, { ...options, method: 'DELETE' }),
};

  postgres:
    image: postgres:16-alpine
    container_name: cmep-postgres
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-cmep}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?POSTGRES_PASSWORD musi byt nastaveny}
      POSTGRES_DB: ${POSTGRES_DB:-cmep}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-cmep} -d ${POSTGRES_DB:-cmep}"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped
    networks:
      - cmep-network

  redis:
    image: redis:7-alpine
    container_name: cmep-redis
    volumes:
      - redis_data:/data
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5
    restart: unless-stopped
    networks:
      - cmep-network

  api:
    build:
      context: .
      dockerfile: app/Dockerfile
    container_name: cmep-api
    environment:
      DATABASE_URL: postgresql+asyncpg://${POSTGRES_USER:-cmep}:${POSTGRES_PASSWORD:?POSTGRES_PASSWORD musi byt nastaveny}@postgres:5432/${POSTGRES_DB:-cmep}
      REDIS_HOST: redis
      JWT_SECRET_KEY: ${JWT_SECRET_KEY:?JWT_SECRET_KEY musi byt nastaveny}
      JWT_REFRESH_SECRET_KEY: ${JWT_REFRESH_SECRET_KEY:?JWT_REFRESH_SECRET_KEY musi byt nastaveny}
      CORS_ALLOWED_ORIGINS: ${CORS_ALLOWED_ORIGINS:-http://localhost:3001}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    restart: unless-stopped
    networks:
      - cmep-network

  web:
    build:
      context: ./apps/web
      dockerfile: Dockerfile
    container_name: cmep-web
    environment:
      NEXT_PUBLIC_API_BASE_URL: ${NEXT_PUBLIC_API_BASE_URL:-http://localhost:8000}
    depends_on:
      api:
        condition: service_started
    restart: unless-stopped
    networks:
      - cmep-network

networks:
  cmep-network:
    driver: bridge

volumes:
  postgres_data:
  redis_data:

===== start-platform.sh =====
#!/data/data/com.termux/files/usr/bin/bash
set -euo pipefail
ROOT="$HOME/waste-platform"
cd "$ROOT"

echo "=== CMEP start ==="

# Postgres
if ! pg_isready -h localhost -p 5432 >/dev/null 2>&1; then
  echo "[pg] starting..."
  pg_ctl -D "$PREFIX/var/lib/postgresql" start || true
  sleep 2
fi
pg_isready -h localhost -p 5432 && echo "[pg] ok" || echo "[pg] FAIL"

# Redis
if ! redis-cli ping 2>/dev/null | grep -q PONG; then
  echo "[redis] starting..."
  redis-server --daemonize yes || true
  sleep 1
fi
redis-cli ping 2>/dev/null | grep -q PONG && echo "[redis] ok" || echo "[redis] FAIL"

# stop old
pkill -f "uvicorn app.main:app" 2>/dev/null || true
pkill -f "next dev" 2>/dev/null || true
sleep 1

# backend
echo "[backend] :8001"
nohup uvicorn app.main:app --host 0.0.0.0 --port 8001 --reload \
  > "$HOME/cmep-backend.log" 2>&1 &
BACKEND_PID=$!
echo "backend pid $BACKEND_PID"

# frontend
echo "[frontend] :3001"
cd "$ROOT/apps/web"
nohup npm run dev -- --port 3001 --hostname 0.0.0.0 \
  > "$HOME/cmep-frontend.log" 2>&1 &
FRONTEND_PID=$!
echo "frontend pid $FRONTEND_PID"

# Počkaj 5 sekúnd na naštartovanie
sleep 5

# Skontroluj backend (stačí raz)
if curl -s -m 5 http://localhost:8001/health >/dev/null 2>&1; then
  echo "✅ backend ready"
else
  echo "❌ backend not responding (check ~/cmep-backend.log)"
fi

# Skontroluj frontend – opakovane až 30 pokusov
echo "Čakám na frontend..."
for i in {1..30}; do
  if curl -s -o /dev/null -w "%{http_code}" http://localhost:3001 | grep -q "^2"; then
    echo "✅ frontend ready (HTTP 200)"
    break
  fi
  if [ $i -eq 30 ]; then
    echo "❌ frontend not ready after 30s (check ~/cmep-frontend.log)"
  fi
  sleep 1
done

echo ""
echo "API:  http://localhost:8001/docs"
echo "WEB:  http://localhost:3001"
echo "Logy: tail -f ~/cmep-backend.log ~/cmep-frontend.log"

def _is_uuidish(type_obj) -> bool:
    if type_obj is None:
        return False
    name = type(type_obj).__name__.lower()
    text = str(type_obj).lower()
    return "uuid" in name or "uuid" in text


def _is_sqlite_numericish(type_obj) -> bool:
    if type_obj is None:
        return False
    name = type(type_obj).__name__.lower()
    text = str(type_obj).lower()
    return "numeric" in name or "numeric" in text


def custom_compare_type(context, inspected_column, metadata_column, inspected_type, metadata_type):
    dialect_name = context.dialect.name

    if dialect_name == "sqlite":
        if _is_sqlite_numericish(inspected_type) and _is_uuidish(metadata_type):
            return False

    return None


def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=custom_compare_type,
        compare_server_default=True,
    )
    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(
        connection=connection,
        target_metadata=target_metadata,
        compare_type=custom_compare_type,
        compare_server_default=True,
    )
    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)
    await connectable.dispose()


def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

from __future__ import annotations

import importlib
import logging
from contextlib import asynccontextmanager

from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware

from app.core.config import settings
from app.core.http_exception_handlers import register_http_exception_handlers

logger = logging.getLogger("cmep.main")


@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info(
        "Starting CMEP app env=%s docs=%s db=%s",
        settings.app_env,
        settings.ENABLE_PUBLIC_DOCS,
        settings.DATABASE_URL,
    )
    yield
    logger.info("Stopping CMEP app")


def _cors_origins() -> list[str]:
    if isinstance(settings.CORS_ORIGINS, list):
        return settings.CORS_ORIGINS
    if isinstance(settings.CORS_ORIGINS, str) and settings.CORS_ORIGINS.strip():
        return [settings.CORS_ORIGINS.strip()]
    return ["http://127.0.0.1:3001", "http://localhost:3001"]


def create_app() -> FastAPI:
    app = FastAPI(
        title=settings.app_name,
        description=settings.app_description,
        version=settings.app_version,
        docs_url="/docs" if settings.ENABLE_PUBLIC_DOCS else None,
        redoc_url="/redoc" if settings.ENABLE_PUBLIC_DOCS else None,
        openapi_url="/openapi.json" if settings.ENABLE_PUBLIC_DOCS else None,
        root_path=settings.ROOT_PATH or "",
        lifespan=lifespan,
    )

    register_http_exception_handlers(app)

    app.add_middleware(
        CORSMiddleware,
        allow_origins=_cors_origins(),
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    mounted: list[str] = []
    skipped: list[dict[str, str]] = []

    @app.get("/health")
    async def health():
        return {
            "status": "ok",
            "service": settings.app_name,
            "version": settings.app_version,
            "env": settings.app_env,
            "database_url": settings.DATABASE_URL,
            "mounted_routers": mounted,
            "skipped_routers": skipped,
        }

    try:
        from app.shared.presentation.api_router import api_router

        pref = getattr(api_router, "prefix", "") or ""
        if pref.startswith("/api"):
            app.include_router(api_router)
        else:
            app.include_router(api_router, prefix="/api/v1")
        mounted.append("shared.api_router")
    except Exception as e:
        logger.exception("Failed to include shared.api_router")
        skipped.append({"module": "shared.api_router", "error": str(e)})

    module_routers = [
        "app.modules.auth.presentation.router",
        "app.modules.users.presentation.router",
        "app.modules.companies.presentation.router",
        "app.modules.company_memberships.presentation.router",
        "app.modules.listings.presentation.router",
        "app.modules.inquiries.presentation.router",
        "app.modules.documents.presentation.router",
        "app.modules.compliance.presentation.router",
        "app.modules.notifications.presentation.router",
        "app.modules.analytics.presentation.router",
        "app.modules.admin.presentation.router",
        "app.modules.uploads.presentation.router",
    ]

    for mod_path in module_routers:
        tag = mod_path.split(".")[2]
        try:
            mod = importlib.import_module(mod_path)
            router = getattr(mod, "router", None)
            if router is None:
                skipped.append({"module": tag, "error": "router attribute missing"})
                logger.warning("Router missing for module=%s", tag)
                continue

            app.include_router(router, prefix="/api/v1")
            mounted.append(tag)
        except Exception as ex:
            logger.exception("Failed to include router module=%s", tag)
            skipped.append({"module": tag, "error": str(ex)})

    app.state.mounted_routers = mounted
    app.state.skipped_routers = skipped

    logger.info("Mounted routers=%s skipped=%s", mounted, skipped)

    return app


app = create_app()

import json
from typing import List, Union

from pydantic import BaseModel, Field, field_validator
from pydantic_settings import BaseSettings, SettingsConfigDict


SQLITE_DEFAULT = "sqlite+aiosqlite:///./var/cmep.db"

    echo: bool = False
    pool_size: int = 5
    pool_pre_ping: bool = True

    def model_post_init(self, __context) -> None:

def _is_placeholder_db_value(raw: str | None) -> bool:
    value = str(raw or "").strip()
    return (
        not value
        or "{" in value
        or "}" in value
        or "POSTGRES_" in value
        or value.lower() in {"none", "null", "undefined"}
    )


def _normalize_database_url(raw: str) -> str:
    value = raw.strip()

    if value.startswith("postgresql://"):
        return value.replace("postgresql://", "postgresql+asyncpg://", 1)

    if value.startswith("postgres://"):
        return value.replace("postgres://", "postgresql+asyncpg://", 1)

    if value.startswith("sqlite:///"):
        return value.replace("sqlite:///", "sqlite+aiosqlite:///", 1)

    return value


class Settings(BaseSettings):
    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        extra="ignore",
        case_sensitive=False,
        populate_by_name=True,
    )

    app_name: str = "CMEP Platform"
    app_version: str = "0.1.0"
    app_description: str = "CMEP - Platform for circular materials exchange"

    app_env: str = Field(default="development", alias="APP_ENV")
    DEBUG: bool = True
    ENABLE_PUBLIC_DOCS: bool = True
    ROOT_PATH: str = ""

    HOST: str = "0.0.0.0"
    PORT: int = 8001

    CORS_ORIGINS: Union[List[str], str] = Field(
        default_factory=lambda: [
            "http://127.0.0.1:3001",
            "http://localhost:3001",
            "http://localhost:3000",
            "http://127.0.0.1:3000",
        ]
    )

    DATABASE_URL: str = SQLITE_DEFAULT

    JWT_SECRET_KEY: str = "cmep-dev-change-me-to-at-least-32-chars-long-secret"
    JWT_ALGORITHM: str = "HS256"
    ACCESS_TOKEN_EXPIRE_MINUTES: int = 60
    REFRESH_TOKEN_EXPIRE_DAYS: int = 14

    jwt: JWTSettings | None = None
    redis: RedisSettings = Field(default_factory=RedisSettings)
    rate_limit: RateLimitSettings = Field(default_factory=RateLimitSettings)
    database: DatabaseSettings | None = None

    GOOGLE_CLIENT_ID: str = ""
    GOOGLE_CLIENT_SECRET: str = ""
    GOOGLE_REDIRECT_URI: str = "http://127.0.0.1:3001/api/auth/callback/google"
    NEXT_PUBLIC_API_URL: str = "http://127.0.0.1:8001"
    NEXT_PUBLIC_GOOGLE_CLIENT_ID: str = ""

    @property
    def is_development(self) -> bool:
        return self.app_env.lower() in {"development", "dev", "local", "test"}

    @field_validator("CORS_ORIGINS", mode="before")
    @classmethod
    def parse_cors(cls, value):
        if value is None or value == "":
            return ["http://127.0.0.1:3001", "http://localhost:3001"]
        if isinstance(value, list):
            return value

        text = str(value).strip()
        if text.startswith("["):
            return json.loads(text)

        return [item.strip() for item in text.split(",") if item.strip()]

    def model_post_init(self, __context) -> None:
        raw_db = str(self.DATABASE_URL or "").strip()

        if _is_placeholder_db_value(raw_db):
            if self.is_development:
                database_url = SQLITE_DEFAULT
            else:
                raise ValueError("DATABASE_URL is missing or invalid for non-development environment.")
        else:
            database_url = _normalize_database_url(raw_db)

        object.__setattr__(self, "DATABASE_URL", database_url)
        object.__setattr__(
            self,
            "database",
            DatabaseSettings(
                url=database_url,
                async_url=database_url,
            ),
        )

        if self.jwt is None:
            object.__setattr__(
                self,
                "jwt",
                JWTSettings(
                    secret_key=self.JWT_SECRET_KEY,
                    secret=self.JWT_SECRET_KEY,
                    algorithm=self.JWT_ALGORITHM,
                    access_token_expire_minutes=self.ACCESS_TOKEN_EXPIRE_MINUTES,
                    expire_minutes=self.ACCESS_TOKEN_EXPIRE_MINUTES,
                    refresh_token_expire_days=self.REFRESH_TOKEN_EXPIRE_DAYS,
                ),
            )


settings = Settings()

import { tokenStore } from '../auth/token-store';

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const baseUrl = process.env.NEXT_PUBLIC_API_BASE_URL || 'http://localhost:8001';
  const url = `${baseUrl}${path.startsWith('/') ? path : '/' + path}`;
  const token = tokenStore.getToken();

  const headers: HeadersInit = {
    'Content-Type': 'application/json',
    ...(options.headers || {}),
  };

  if (token) {
    headers['Authorization'] = `Bearer ${token}`;
  }

  const response = await fetch(url, { ...options, headers });

  if (!response.ok) {
    if (response.status === 401) {
      tokenStore.clearToken();
      if (typeof window !== 'undefined') window.location.href = '/login';
    }
    const errorData = await response.json().catch(() => ({}));
    throw {
      message: errorData.detail || errorData.message || 'An unexpected error occurred',
      status: response.status,
      errors: errorData.errors,
    };
  }
  if (response.status === 204) return {} as T;
  return response.json();
}

export const http = {
  get: <T>(path: string, options?: RequestInit) => request<T>(path, { ...options, method: 'GET' }),
  post: <T>(path: string, body?: any, options?: RequestInit) => request<T>(path, { ...options, method: 'POST', body: body ? JSON.stringify(body) : undefined }),
  put: <T>(path: string, body?: any, options?: RequestInit) => request<T>(path, { ...options, method: 'PUT', body: body ? JSON.stringify(body) : undefined }),
  patch: <T>(path: string, body?: any, options?: RequestInit) => request<T>(path, { ...options, method: 'PATCH', body: body ? JSON.stringify(body) : undefined }),
  delete: <T>(path: string, options?: RequestInit) => request<T>(path, { ...options, method: 'DELETE' }),
};

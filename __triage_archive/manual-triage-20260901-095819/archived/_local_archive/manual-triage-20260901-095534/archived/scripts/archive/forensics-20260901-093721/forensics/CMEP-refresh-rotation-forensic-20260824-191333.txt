def _decode_payload(token: str) -> dict:
    payload = token.split(".")[1]
    payload += "=" * (-len(payload) % 4)
    return json.loads(base64.urlsafe_b64decode(payload.encode()).decode())


@pytest.mark.asyncio
async def test_password_reset_revokes_old_refresh_token():
    email = f"pwreset-{uuid.uuid4().hex[:8]}@example.com"
    password = "StartPass123!"
    new_password = "NewPass123!"
    now = datetime.now(UTC)
    user_id = str(uuid.uuid4())

    password_service = PasswordService()

    async with SessionLocal() as session:
        user = UserModel(
            id=user_id,
            status="active",
            email=email,
            password_hash=password_service.hash_password(password),
            first_name="Test",
            last_name="User",
            display_name="Test User",
            phone=None,
            locale="sk",
            timezone="Europe/Bratislava",
            last_login_at=None,
            created_at=now,
            created_by_user_id=None,
            updated_at=now,
            updated_by_user_id=None,
        )
        session.add(user)
        await session.commit()

    async with AsyncClient(
        transport=ASGITransport(app=app),
        base_url="http://test",
    ) as client:
        login = await client.post(
            "/api/v1/auth/login",
            json={"email": email, "password": password},
        )
        assert login.status_code == 200, login.text
        login_data = login.json()

        assert "accessToken" in login_data
        assert "refreshToken" in login_data

        access_payload = _decode_payload(login_data["accessToken"])
        refresh_payload = _decode_payload(login_data["refreshToken"])

        assert access_payload["type"] == "access"
        assert refresh_payload["type"] == "refresh"
        assert "iat" in access_payload
        assert "iat" in refresh_payload

        forgot = await client.post(
            "/api/v1/auth/forgot-password",
            json={"email": email},
        )
        assert forgot.status_code == 200, forgot.text
        forgot_data = forgot.json()

        assert forgot_data["ok"] is True
        assert "resetToken" in forgot_data

        reset = await client.post(
            "/api/v1/auth/reset-password",
            json={
                "token": forgot_data["resetToken"],
                "newPassword": new_password,
            },
        )
        assert reset.status_code == 200, reset.text

        old_refresh = await client.post(
            "/api/v1/auth/refresh",
            json={"refreshToken": login_data["refreshToken"]},
        )
        assert old_refresh.status_code == 401, old_refresh.text

        reuse_reset = await client.post(
            "/api/v1/auth/reset-password",
            json={
                "token": forgot_data["resetToken"],
                "newPassword": "AnotherPass123!",
            },
        )
        assert reuse_reset.status_code in (400, 401, 422), reuse_reset.text

        login_old_password = await client.post(
            "/api/v1/auth/login",
            json={"email": email, "password": password},
        )
        assert login_old_password.status_code in (400, 401), login_old_password.text

        login_new_password = await client.post(
            "/api/v1/auth/login",
            json={"email": email, "password": new_password},
        )
        assert login_new_password.status_code == 200, login_new_password.text

    49	            not getattr(settings.redis, "enabled", False)
    50	            or self.redis_url.startswith("memory:")
    51	            or self.redis_url in ("", "none", "disabled")
    52	        )

    54	    async def _get_client(self) -> Any:
    55	        if self._client is not None:
    56	            return self._client
    57	        if self._use_memory:
    58	            self._client = _MemoryBackend()
    59	            return self._client
    60	        try:
    61	            import redis.asyncio as redis
    62	            self._client = redis.from_url(self.redis_url, decode_responses=True)
    63	            return self._client
    64	        except Exception:
    65	            self._client = _MemoryBackend()
    66	            self._use_memory = True
    67	            return self._client

    71	        ttl = int((exp - datetime.now(timezone.utc)).total_seconds())

    83	        client = await self._get_client()

    86	        await client.set(

    93	        client = await self._get_client()
    94	        value = await client.get(f"revoked:user:{subject}:after")
    95	        if not value:
    96	            return None
    97	        try:
    98	            return datetime.fromisoformat(value)
    99	        except Exception:
   100	            return None

    29	            RefreshSessionModel.revoked_at.is_(None),

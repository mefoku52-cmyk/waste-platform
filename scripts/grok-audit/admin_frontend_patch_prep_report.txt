     4	async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
     5	  const baseUrl = config.apiBaseUrl;
     6	  const url = `${baseUrl}${path.startsWith('/') ? path : `/${path}`}`;
     7	  const token = tokenStore.getToken();
     8	
     9	  const headers: Record<string, string> = {
    10	    'Content-Type': 'application/json',
    11	    ...(options.headers as Record<string, string> | undefined),
    12	  };
    13	
    14	  if (token) {
    15	    headers.Authorization = `Bearer ${token}`;
    16	  }
    17	
    18	  const response = await fetch(url, {
    19	    ...options,
    20	    headers,
    21	  });
    22	
    23	  if (!response.ok) {
    24	    if (response.status === 401) {
    25	      tokenStore.clearToken();
    26	      if (typeof window !== 'undefined') {
    27	        window.location.href = '/login';
    28	      }
    29	    }
    30	
    31	    const errorData = await response.json().catch(() => ({}));
    32	    throw {
    33	      message: errorData.detail ?? errorData.message ?? 'An unexpected error occurred',
    34	      status: response.status,
    35	      errors: errorData.errors,
    36	    };
    37	  }
    38	
    39	  if (response.status === 204) {
    40	    return undefined as T;
    41	  }
    42	
    43	  return response.json();
    44	}

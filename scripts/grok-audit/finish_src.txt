def _is_placeholder_db_value(raw: str | None) -> bool:
    value = str(raw or "").strip()
    return (
        not value
        or "{" in value
        or "}" in value
        or "POSTGRES_" in value
        or value.lower() in {"none", "null", "undefined"}
    )


def _normalize_database_url(raw: str) -> str:
    value = raw.strip()

    if value.startswith("postgresql://"):
        return value.replace("postgresql://", "postgresql+asyncpg://", 1)

    if value.startswith("postgres://"):
        return value.replace("postgres://", "postgresql+asyncpg://", 1)

    if value.startswith("sqlite:///"):
        return value.replace("sqlite:///", "sqlite+aiosqlite:///", 1)

    return value

from app.core.config import settings
from app.core.http_exception_handlers import register_http_exception_handlers

logger = logging.getLogger("cmep.main")


@asynccontextmanager
async def lifespan(app: FastAPI):
    logger.info(
        "Starting CMEP app env=%s docs=%s db=%s",
        settings.app_env,
        settings.ENABLE_PUBLIC_DOCS,
        settings.DATABASE_URL,
    )
    yield
    logger.info("Stopping CMEP app")


def _cors_origins() -> list[str]:
    if isinstance(settings.CORS_ORIGINS, list):
        return settings.CORS_ORIGINS
    if isinstance(settings.CORS_ORIGINS, str) and settings.CORS_ORIGINS.strip():
        return [settings.CORS_ORIGINS.strip()]
    return ["http://127.0.0.1:3001", "http://localhost:3001"]


def create_app() -> FastAPI:
    app = FastAPI(
        title=settings.app_name,
        description=settings.app_description,
        version=settings.app_version,
        docs_url="/docs" if settings.ENABLE_PUBLIC_DOCS else None,
        redoc_url="/redoc" if settings.ENABLE_PUBLIC_DOCS else None,
        openapi_url="/openapi.json" if settings.ENABLE_PUBLIC_DOCS else None,
        root_path=settings.ROOT_PATH or "",
        lifespan=lifespan,
    )

    register_http_exception_handlers(app)

    app.add_middleware(
        CORSMiddleware,
        allow_origins=_cors_origins(),
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )

    mounted: list[str] = []
    skipped: list[dict[str, str]] = []

    @app.get("/health")
    async def health():
        return {
            "status": "ok",
            "service": settings.app_name,
            "version": settings.app_version,
            "env": settings.app_env,
            "database_url": settings.DATABASE_URL,
            "mounted_routers": mounted,
            "skipped_routers": skipped,
        }

    try:
        from app.shared.presentation.api_router import api_router

        pref = getattr(api_router, "prefix", "") or ""
        if pref.startswith("/api"):
            app.include_router(api_router)
        else:
            app.include_router(api_router, prefix="/api/v1")
        mounted.append("shared.api_router")
    except Exception as e:
        logger.exception("Failed to include shared.api_router")
        skipped.append({"module": "shared.api_router", "error": str(e)})

    module_routers = [
        "app.modules.auth.presentation.router",
        "app.modules.users.presentation.router",
        "app.modules.companies.presentation.router",
        "app.modules.company_memberships.presentation.router",
        "app.modules.listings.presentation.router",
        "app.modules.inquiries.presentation.router",
        "app.modules.documents.presentation.router",
        "app.modules.compliance.presentation.router",
        "app.modules.notifications.presentation.router",
        "app.modules.analytics.presentation.router",
        "app.modules.admin.presentation.router",
        "app.modules.uploads.presentation.router",
    ]

    for mod_path in module_routers:
        tag = mod_path.split(".")[2]
        try:
            mod = importlib.import_module(mod_path)
            router = getattr(mod, "router", None)
            if router is None:
                skipped.append({"module": tag, "error": "router attribute missing"})
                logger.warning("Router missing for module=%s", tag)
                continue

            app.include_router(router, prefix="/api/v1")
            mounted.append(tag)
        except Exception as ex:
            logger.exception("Failed to include router module=%s", tag)
            skipped.append({"module": tag, "error": str(ex)})

    app.state.mounted_routers = mounted
    app.state.skipped_routers = skipped

    logger.info("Mounted routers=%s skipped=%s", mounted, skipped)

    return app


app = create_app()

        self.redis = redis_client

    def _get_redis(self):
        if self.redis is not None:
            return self.redis
        try:
            import redis

            host = getattr(settings.redis, "host", "localhost")
            port = getattr(settings.redis, "port", 6379)
            url = getattr(settings.redis, "url", None) or f"redis://{host}:{port}/0"
            self.redis = redis.from_url(url, decode_responses=True)
            self.redis.ping()
            return self.redis
        except Exception:
            self.redis = None
            return None

    def _store_token(self, token: str, user_id: str) -> None:
        from app.shared.domain.exceptions import ValidationDomainError

        key = RESET_PREFIX + _token_digest(token)
        r = self._get_redis()
        if r is not None:
            r.set(key, user_id, ex=RESET_TTL_SECONDS)
            return

        if settings.is_development:
            logger.warning("Password reset using in-memory fallback storage")
            PasswordResetService._mem[key] = user_id
            return

        logger.error("Password reset unavailable: Redis is not reachable")
        raise ValidationDomainError(
            "Obnova hesla je dočasne nedostupná. Skúste to neskôr."
        )

    def _pop_token(self, token: str) -> str | None:
        key = RESET_PREFIX + _token_digest(token)
        r = self._get_redis()
        if r is not None:
            uid = r.get(key)
            if uid:
                r.delete(key)
            return uid

        if settings.is_development:
            return PasswordResetService._mem.pop(key, None)

        return None

    async def request_reset(self, email: str) -> dict:
        email_n = email.lower().strip()
        user = await self.user_repo.get_by_email(email_n)
        result = {
            "ok": True,
            "message": "Ak účet existuje, poslali sme inštrukcie na e-mail.",
        }
        if user is None:
            return result

        token = secrets.token_urlsafe(32)
        self._store_token(token, str(user.id))

        base = settings.frontend_url.rstrip("/")
        nl = chr(10)
        body = (
            "Obnova hesla CMEP"
            + nl + nl
            + f"Token (platný 30 min): {token}"
            + nl + nl
            + f"Link: {base}/reset-password?token={token}"
            + nl
        )
        sent = send_email(to=email_n, subject="CMEP – obnova hesla", body=body)
        result["emailSent"] = bool(sent)
        if settings.is_development:
            result["resetToken"] = token
        return result

    async def reset_password(self, token: str, new_password: str) -> None:
        from app.shared.domain.exceptions import (
            NotFoundDomainError,
            ValidationDomainError,
        )

        if not token or len(new_password) < 8:
            raise ValidationDomainError(
                "Neplatný token alebo heslo (min. 8 znakov)."
            )
        user_id = self._pop_token(token)
        if not user_id:
            raise ValidationDomainError("Token je neplatný alebo expirovaný.")
        user = await self.user_repo.get_by_id(UUID(str(user_id)))
        if user is None:
            raise NotFoundDomainError("User not found")
        user.password_hash = self.password_service.hash_password(new_password)
        await self.user_repo.update(user)
        await self.jwt_service.revoke_all_user_tokens(str(user.id))

def _is_placeholder_db_value(raw: str | None) -> bool:
    value = str(raw or "").strip()
    return (
        not value
        or "{" in value
        or "}" in value
        or "POSTGRES_" in value
        or value.lower() in {"none", "null", "undefined"}
    )


def _normalize_database_url(raw: str) -> str:
    value = raw.strip()

    if value.startswith("postgresql://"):
        return value.replace("postgresql://", "postgresql+asyncpg://", 1)

    if value.startswith("postgres://"):
        return value.replace("postgres://", "postgresql+asyncpg://", 1)

    if value.startswith("sqlite:///"):
        return value.replace("sqlite:///", "sqlite+aiosqlite:///", 1)

    return value

"""Best-effort SMTP sender. Without credentials logs and returns False."""

from __future__ import annotations

import logging
import smtplib
from email.message import EmailMessage

from app.core.config import settings

logger = logging.getLogger(__name__)


def send_email(*, to: str, subject: str, body: str) -> bool:
    host = getattr(settings, "SMTP_HOST", "")
    port = int(getattr(settings, "SMTP_PORT", 587))
    user = getattr(settings, "SMTP_USERNAME", "")
    password = getattr(settings, "SMTP_PASSWORD", "")
    from_address = getattr(settings, "EMAIL_FROM", "") or user
    use_tls = bool(getattr(settings, "SMTP_USE_TLS", True))

    if not host or not user or not password:
        logger.warning(
            "EMAIL not configured – message to %s not sent. subject=%s",
            to,
            subject,
        )
        logger.info("EMAIL_DEV_BODY to=%s\n%s", to, body)
        return False

    msg = EmailMessage()
    msg["From"] = from_address
    msg["To"] = to
    msg["Subject"] = subject
    msg.set_content(body)

    try:
        with smtplib.SMTP(host, port, timeout=20) as smtp:
            if use_tls:
                smtp.starttls()
            smtp.login(user, password)
            smtp.send_message(msg)
        return True
    except Exception as exc:
        logger.exception("send_email failed: %s", exc)
        return False
